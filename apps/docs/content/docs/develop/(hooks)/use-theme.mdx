## Example

### Building a Theme Switch Component

The `useTheme` hook provides an easy way to manage theme states within your application. Below, we define a custom component that allows users to toggle between light and dark modes.

<Tabs items={["Toggle", "Select"]}>
  <Tab value="Toggle" className="not-prose">
    ```tsx title="theme-toggle.tsx"
    import { useTheme } from "@amerfathullah/myds-react/hooks";

    const ThemeToggle = () => {
      const { theme, setTheme } = useTheme();

      return (
        <button onClick={() => setTheme(theme === "light" ? "dark" : "light")}>
          Switch to {theme === "light" ? "dark" : "light"} mode
        </button>
      );
    };

    export { ThemeToggle }
    ```

  </Tab>
  <Tab value="Select">
    ```tsx title="theme-select.tsx"
    import { useTheme } from "@amerfathullah/myds-react/hooks";
    import {
      Select,
      SelectTrigger,
      SelectValue,
      SelectContent,
      SelectItem,
    } from "@amerfathullah/myds-react/select";

    const ThemeSelect = () => {
      const { theme, setTheme } = useTheme();

      const handleChange = (newTheme: string) => {
        setTheme(newTheme);
      };

      return (
        <Select
          size="small"
          variant="outline"
          value={theme}
          onValueChange={handleChange}
        >
          <SelectTrigger aria-label="Theme">
            <SelectValue label="Theme" />
          </SelectTrigger>
          <SelectContent>
            <SelectItem value="light">Light</SelectItem>
            <SelectItem value="dark">Dark</SelectItem>
          </SelectContent>
        </Select>
      );
    };

    export { ThemeSelect }
    ```

  </Tab>
</Tabs>

### Declaring a Globally Accessible ThemeProvider

To apply the theme context across the application, wrap the entire layout with the `ThemeProvider`. This ensures that all child components can access the theme state and toggle functionality. In a Next.js App Router app, `ThemeProvider` and `useTheme` work only in client components, so the layout starts with `"use client"`.

```tsx title="layout.tsx"
"use client";

import type { ReactNode } from "react";
import { ThemeProvider } from "@amerfathullah/myds-react/hooks";
import { ThemeToggle } from "./theme-toggle";

export default function Layout({ children }: { children: ReactNode }) {
  return (
    <ThemeProvider>
      <div className="app-container">
        <ThemeToggle />
        {children}
      </div>
    </ThemeProvider>
  );
}
```
